local function makeDraggable(frame, handle)
    local dragging, dragStart, startPos = false, nil, nil
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            dragStart = input.Position
            startPos  = frame.Position
        end
    end)
    handle.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (
            input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch
        ) then
            local d = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)
end

local duelBases   = {}

local function IsPlayerInDuel(player)
    if not player then return false end
    return player:GetAttribute("__duels_block_steal") == true
end

local function DetectDuelBases()
    duelBases   = {}
    duelPlayers = {}
    local Plots = workspace:FindFirstChild("Plots")
    if not Plots then return end
    for _, plot in ipairs(Plots:GetChildren()) do
        local sign = plot:FindFirstChild("PlotSign")
        if not sign then continue end
        local sg = sign:FindFirstChild("SurfaceGui")
        local fr = sg and sg:FindFirstChild("Frame")
        local tl = fr and fr:FindFirstChild("TextLabel")
        local baseText = tl and tl.Text or nil
        if not baseText or baseText == "Empty Base" then continue end
        local nickname = baseText:match("^(.-)'") or baseText
        local ownerPlayer = nil
        for _, p in ipairs(Players:GetPlayers()) do
            if p.DisplayName == nickname or p.Name == nickname then
                ownerPlayer = p; break
            end
        end
        if ownerPlayer and IsPlayerInDuel(ownerPlayer) then
            duelBases[plot.Name]            = {plot = plot, owner = ownerPlayer, ownerName = ownerPlayer.Name}
            duelPlayers[ownerPlayer.UserId] = true
        end
    end
end

local function parseMoneyPerSec(text)
    if not text or not text:find("/s") then return 0 end
    text = text:gsub("%$", ""):gsub("/s", ""):gsub("%s+", "")
    local mult = 1
    if text:find("[Tt]") then mult = 1e12
    elseif text:find("[Bb]") then mult = 1e9
    elseif text:find("[Mm]") then mult = 1e6
    elseif text:find("[Kk]") then mult = 1e3
    end
    text = text:gsub("[KkMmBbTt]", "")
    return (tonumber(text) or 0) * mult
end

local function getUsernameFromDisplayName(dn)
    for _, p in pairs(Players:GetPlayers()) do
        if p.DisplayName == dn or p.Name == dn then return p.Name end
    end
    return nil
end

local function getUsernameFromPlot(plot)
    local ok, r = pcall(function()
        if plot:FindFirstChild("Owner") then
            local o = plot.Owner
            if o:IsA("StringValue") then return o.Value end
            if o:IsA("ObjectValue") and o.Value then return o.Value.Name end
        end
        return plot:GetAttribute("Owner") or plot:GetAttribute("OwnerName") or nil
    end)
    return ok and r or nil
end

local function getPlotOwnerInfo(plotId)
    local ok, r = pcall(function()
        local plot = workspace.Plots:FindFirstChild(plotId)
        if not plot then return nil end
        local sign = plot:FindFirstChild("PlotSign")
        if sign then
            local sg = sign:FindFirstChild("SurfaceGui")
            local fr = sg and sg:FindFirstChild("Frame")
            local tl = fr and fr:FindFirstChild("TextLabel")
            if tl and tl.Text ~= "" then
                local dn = tl.Text:gsub("'s [Bb]ase", ""):gsub("'s [Pp]lot", ""):match("^%s*(.-)%s*$")
                if dn ~= "" then return getUsernameFromDisplayName(dn) or getUsernameFromPlot(plot) or dn end
            end
        end
        return getUsernameFromPlot(plot)
    end)
    return ok and r or nil
end

local function findOwnerForBrainrot(obj)
    local cur = obj.Parent
    while cur and cur ~= workspace do
        if cur.Parent and cur.Parent.Name == "Plots" then
            return getPlotOwnerInfo(cur.Name) or "Unknown"
        end
        cur = cur.Parent
    end
    return "Unknown"
end

local function findPlotFolderForBrainrot(obj)
    local cur = obj.Parent
    while cur and cur ~= workspace do
        if cur.Parent and cur.Parent.Name == "Plots" then return cur end
        cur = cur.Parent
    end
    return nil
end

local function getPlotOwnerName(plotFolder)
    local ok, result = pcall(function()
        local sign = plotFolder:FindFirstChild("PlotSign")
        if not sign then return nil end
        local sg = sign:FindFirstChild("SurfaceGui")
        local fr = sg and sg:FindFirstChild("Frame")
        local tl = fr and fr:FindFirstChild("TextLabel")
        if not tl or tl.Text == "" then return nil end
        local nickname = tl.Text:match("^(.-)'") or tl.Text
        for _, p in ipairs(Players:GetPlayers()) do
            if p.DisplayName == nickname or p.Name == nickname then return p.Name end
        end
        return nickname
    end)
    return ok and result or nil
end

local function isPlotInDuel(plotFolder)
    if not plotFolder then return false end
    if duelBases[plotFolder.Name] then return true end
    local ownerName = getPlotOwnerName(plotFolder)
    if not ownerName then return false end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name == ownerName or p.DisplayName == ownerName then
            return duelPlayers[p.UserId] == true or IsPlayerInDuel(p)
        end
    end
    return false
end

local MIN_GENERATION = 10

local JUNK_ANCESTORS = {
    AnimalOverhead   = true,
    Overhead         = true,
    NameTag          = true,
    HumanoidRootPart = true,
    Head             = true,
}
local JUNK_NAMES = {
    AnimalOverhead = true, Base = true, Spawn = true,
    Model = true, Part = true, Handle = true,
}

local function isInsideJunkAncestor(obj)
    local cur = obj.Parent
    while cur and cur ~= workspace do
        if JUNK_ANCESTORS[cur.Name] then return true end
        cur = cur.Parent
    end
    return false
end

local function scanAllBrainrots()
    local detections = {}
    local descendants = workspace:GetDescendants()
    local batchSize   = 50
    for i = 1, #descendants, batchSize do
        local batchEnd = math.min(i + batchSize - 1, #descendants)
        for j = i, batchEnd do
            local obj = descendants[j]
            if not obj:IsA("TextLabel") then continue end
            if isInsideJunkAncestor(obj) then continue end
            local isGenLabel = (
                obj.Name == "Generation"      or obj.Name == "MoneyPerSec" or
                obj.Name == "Income"          or obj.Name == "GenLabel"    or
                obj.Name == "GenerationLabel" or
                (obj.Text and obj.Text:find("/s") and obj.Text:find("%$"))
            )
            if isGenLabel then
                local moneyValue = parseMoneyPerSec(obj.Text)
                local moneyM     = moneyValue / 1e6
                if moneyM >= MIN_GENERATION then
                    local parent = obj.Parent
                    local name   = "Unknown"
                    if parent then
                        local dn = parent:FindFirstChild("DisplayName")
                        if dn and dn:IsA("TextLabel") and dn.Text ~= "" and not JUNK_NAMES[dn.Text] then
                            name = dn.Text
                        elseif parent:FindFirstChild("BrainrotName") and parent.BrainrotName.Value ~= "" then
                            name = parent.BrainrotName.Value
                        elseif parent:FindFirstChild("Title") and parent.Title:IsA("TextLabel")
                            and parent.Title.Text ~= "" and not JUNK_NAMES[parent.Title.Text] then
                            name = parent.Title.Text
                        elseif not JUNK_NAMES[parent.Name] then
                            name = parent.Name
                        elseif parent.Parent and not JUNK_NAMES[parent.Parent.Name]
                            and parent.Parent ~= workspace then
                            name = parent.Parent.Name
                        end
                    end
                    local rawMut = (parent and parent:FindFirstChild("Mutation")
                        and parent.Mutation.Visible and parent.Mutation.Text) or "None"
                    local mutation = rawMut:gsub("<[^>]+>", ""):gsub("%s+", " "):match("^%s*(.-)%s*$")
                    if mutation:lower():find("yin") and mutation:lower():find("yang") then
                        mutation = "YinYang"
                    end
                    if mutation == "" then mutation = "None" end
                    local owner      = findOwnerForBrainrot(obj)
                    local isDuel     = false
                    local plotFolder = findPlotFolderForBrainrot(obj)
                    if plotFolder then isDuel = isPlotInDuel(plotFolder) end
                    table.insert(detections, {
                        name       = name,
                        generation = obj.Text,
                        genM       = moneyM,
                        mutation   = mutation,
                        owner      = owner,
                        isDuel     = isDuel,
                    })
                end
            end
        end
        if i % (batchSize * 2) == 0 then task.wait(0.01) end
    end
    table.sort(detections, function(a, b) return a.genM > b.genM end)
    local best = nil
    for _, d in ipairs(detections) do
        d.priority = getPriority(d.name)
        if best == nil then
            best = d
        elseif d.priority < best.priority then
            best = d
        elseif d.priority == best.priority and d.genM > best.genM then
            best = d
        end
    end
    return detections, best
end

                line = line .. h.name .. " - " .. h.generation
                if h.owner and h.owner ~= "Unknown" then line = line .. " (@" .. h.owner .. ")" end
                if h.isDuel then line = line .. " [DUEL]" end
                desc = desc .. line .. "\n"
            end
            desc = desc .. "```"
            local embed = {
                title       = "1x " .. mutPrefix .. best.name .. " (" .. best.generation .. ")",
                description = desc,
                color       = 65280,
                footer      = {text = "Irish Notifier | " .. os.date("!%m/%d/%Y %I:%M %p")},
                timestamp   = os.date("!%Y-%m-%dT%H:%M:%SZ"),
            }
            local img = BRAINROT_IMAGES[best.name]
            if img then embed.thumbnail = {url = img} end
            http_request({
                Url     = webhookUrl,
                Method  = "POST",
                Headers = {["Content-Type"] = "application/json"},
                Body    = HttpService:JSONEncode({username = "Irish Notifier", embeds = {embed}}),
            })
            print("IRISH COMBINED: Discord sent -> " .. best.name .. " [" .. label .. "]")
        end)
    end)
end

local function pushToRailway(best, serverId)

    pcall(function()
        http_request({
            Url     = RAILWAY_URL .. "/api/ping",
            Method  = "POST",

local function pushToGist(best, serverId)
    if not best then return end
    pcall(function()
        local content = HttpService:JSONEncode({
            servers = {{
                job_id   = serverId,
                place_id = game.PlaceId,
                name     = best.name,
                money    = string.format("%.2fM", best.genM),
                players  = "?",
                ts       = os.time() * 1000,
            }},
            ts    = os.time() * 1000,
            count = 1,
        })
        http_request({
            Url     = "https://api.github.com/gists/" .. GIST_ID,
            Method  = "PATCH",
            Headers = {
                ["Authorization"] = "token " .. GITHUB_TOKEN,
                ["Accept"]        = "application/vnd.github.v3+json",
                ["Content-Type"]  = "application/json",
            },
            Body = HttpService:JSONEncode({files = {["servers.json"] = {content = content}}}),
        })
        print("IRISH COMBINED: Gist pushed -> " .. best.name)
    end)
end

local function isStuckOnLoading()
    if not game:IsLoaded() then return true end
    if not Players.LocalPlayer then return true end
    if not Players.LocalPlayer.Character then return true end
    if #workspace:GetChildren() < 5 then return true end
    return false
end

local function waitForGameLoad(maxWait)
    maxWait = maxWait or 15
    local t = tick()
    repeat
        if tick() - t > maxWait then return false end
        task.wait(0.1)
    until game:IsLoaded() and Players.LocalPlayer and Players.LocalPlayer.Character
    local t2 = tick()
    repeat
        if tick() - t2 > 5 then return false end
        task.wait(0.1)
    until #workspace:GetChildren() >= 5
    return true
end

local function hopToServer()
    local startJobId = game.JobId
    local hopStart   = tick()
    print("IRISH COMBINED: Hopping from " .. startJobId:sub(1, 8) .. "...")
    while game.JobId == startJobId do
        if tick() - hopStart > 30 then
            warn("IRISH COMBINED: Hop timeout, force rejoining...")
            pcall(function() TeleportService:Teleport(game.PlaceId) end)
            task.wait(2)
            hopStart   = tick()
            startJobId = game.JobId
        end
        local servers = {}
        local ok, result = pcall(function()
            return game:HttpGet("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100")
        end)
        if ok and result then
            pcall(function()
                servers = (HttpService:JSONDecode(result) or {}).data or {}
            end)
        end
        local found = false
        for _, s in ipairs(servers) do
            if s.id and s.id ~= game.JobId and s.playing <= (s.maxPlayers - 2) then
                print("IRISH COMBINED: Teleporting (" .. s.playing .. "/" .. s.maxPlayers .. ")")
                pcall(function() TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id) end)
                found = true; break
            end
        end
        if not found then
            warn("IRISH COMBINED: No free servers, rejoining...")
            pcall(function() TeleportService:Teleport(game.PlaceId) end)
        end
        task.wait(2)
    end
    print("IRISH COMBINED: Server changed -> " .. game.JobId:sub(1, 8))
end

local scanComplete    = false
local hopReady        = false
local currentServerId = nil
local lastSentKey     = ""
local serverJoinTime  = tick()

local function performScan()
    scanComplete    = false
    hopReady        = false
    currentServerId = game.JobId
    print("IRISH COMBINED: Loading...")
    if not waitForGameLoad(15) then
        print("IRISH COMBINED: Load timeout")
        scanComplete = true; hopReady = true; return
    end
    local jobWait = tick()
    while (game.JobId == "" or #game.JobId < 30) and tick() - jobWait < 10 do
        task.wait(0.1)
    end
    currentServerId = game.JobId
    if not currentServerId or #currentServerId < 30 then
        print("IRISH COMBINED: Invalid server")
        scanComplete = true; hopReady = true; return
    end
    local t = tick()
    repeat
        task.wait(0.2)
        if isStuckOnLoading() then
            print("IRISH COMBINED: Stuck"); scanComplete = true; hopReady = true; return
        end
        if tick() - t > 12 then
            print("IRISH COMBINED: Plots timeout"); scanComplete = true; hopReady = true; return
        end
    until workspace:FindFirstChild("Plots")
    task.wait(0.5)
    print("IRISH COMBINED: Scanning for 35 seconds...")
    local serverSentKeys = {}
    while tick() - serverJoinTime < 35 do
        local allHits, best = scanAllBrainrots()
        if best then
            local sendKey = best.name .. "|" .. currentServerId
            if not serverSentKeys[sendKey] and sendKey ~= lastSentKey then
                serverSentKeys[sendKey] = true
                lastSentKey             = sendKey
                print("IRISH COMBINED: HIT -> " .. best.name .. " " .. best.generation)
                sendDiscord(allHits, best, currentServerId)
                pushToRailway(best, currentServerId)
                pushToGist(best, currentServerId)
                if genv.IrishNotifier_AddRow then
                    local mM  = math.floor(best.genM * 10) / 10
                    local sub = (best.owner ~= "Unknown") and ("@" .. best.owner) or "Scanned"
                    genv.IrishNotifier_AddRow(best.name, sub, mM, game.PlaceId, currentServerId)
                end
            end
        end
        task.wait(0.1)
    end
    print("IRISH COMBINED: 35s done, hopping...")
    scanComplete = true
    hopReady     = true
end

task.spawn(performScan)

local stuckStartTime = nil

task.spawn(function()
    while true do
        repeat task.wait(0.05) until scanComplete and hopReady
        print("IRISH COMBINED: Hopping to next server...")
        hopToServer()
        serverJoinTime = tick()
        task.spawn(performScan)
    end
end)

task.spawn(function()
    while true do
        task.wait(3)
        if not scanComplete then
            if isStuckOnLoading() then
                if not stuckStartTime then
                    stuckStartTime = tick()
                elseif tick() - stuckStartTime > 10 then
                    print("IRISH COMBINED: Load stuck, forcing...")
                    stuckStartTime = nil
                    scanComplete   = true
                    hopReady       = true
                end
            else
                stuckStartTime = nil
            end
        else
            stuckStartTime = nil
        end
    end
end)

TeleportService.TeleportInitFailed:Connect(function()
    print("IRISH COMBINED: Teleport failed, retrying...")
    scanComplete = true; hopReady = true
end)

local function removePlayerESP(player)
    local key = "player__" .. player.Name
    if espBoxes[key] and espBoxes[key].Parent           then espBoxes[key]:Destroy() end
    espBoxes[key] = nil
    if espBillboards[key] and espBillboards[key].Parent then espBillboards[key]:Destroy() end
    espBillboards[key] = nil
    if espHeartbeats[key] then espHeartbeats[key]:Disconnect(); espHeartbeats[key] = nil end
end

local function clearESPByPrefix(prefix)
    for key, bb in pairs(espBillboards) do
        if tostring(key):sub(1, #prefix) == prefix then
            if bb and bb.Parent then bb:Destroy() end
            espBillboards[key] = nil
        end
    end
    if prefix == "player__" then
        for key, box in pairs(espBoxes) do
            if tostring(key):sub(1, 9) == "player__" then
                if box and box.Parent then box:Destroy() end
                espBoxes[key] = nil
            end
        end
        for key, conn in pairs(espHeartbeats) do
            if tostring(key):sub(1, 9) == "player__" then
                conn:Disconnect(); espHeartbeats[key] = nil
            end
        end
    end
end

local function refreshPlayerESP()
    clearESPByPrefix("player__")
    if not fenv.PlayerESP then return end
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            createPlayerESP(player)
        end
    end
end

local antiLagConn = nil

local function applyAntiLagToObj(obj)
    pcall(function()
        if obj:IsA("ParticleEmitter") or obj:IsA("Trail") then
            obj.Enabled = false
        elseif obj:IsA("Decal") or obj:IsA("Texture") then
            obj.Transparency = 1
        elseif obj:IsA("BasePart") or obj:IsA("MeshPart") then
            obj.Material    = Enum.Material.Plastic
            obj.Reflectance = 0
            obj.CastShadow  = false
        elseif obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles") then
            obj.Enabled = false
        end
    end)
end

local function applyLightingAntiLag(ultra)
    pcall(function()
        Lighting.GlobalShadows            = false
        Lighting.FogEnd                   = 9e9
        Lighting.FogStart                 = 9e9
        Lighting.Brightness               = ultra and 1 or 2
        Lighting.EnvironmentDiffuseScale  = 0
        Lighting.EnvironmentSpecularScale = 0
        if not ultra then Lighting.ClockTime = 12 end
        for _, fx in ipairs(Lighting:GetChildren()) do
            if fx:IsA("PostEffect") or fx:IsA("BloomEffect") or fx:IsA("BlurEffect")
                or fx:IsA("SunRaysEffect") or fx:IsA("ColorCorrectionEffect") then
                fx.Enabled = false
            end
        end
    end)
end

local function enableAntiLag(ultra)
    applyLightingAntiLag(ultra)
    for _, obj in ipairs(Workspace:GetDescendants()) do
        applyAntiLagToObj(obj)
        if ultra then
            pcall(function()
                if obj:IsA("Accessory") or obj:IsA("Clothing")
                    or obj:IsA("Shirt") or obj:IsA("Pants") then
                    local char = LocalPlayer.Character
                    if not (char and obj:IsDescendantOf(char)) then obj:Destroy() end
                end
            end)
        end
    end
    if not antiLagConn then
        antiLagConn = Workspace.DescendantAdded:Connect(applyAntiLagToObj)
    end
    print("[IrishCombined]: Anti-Lag " .. (ultra and "ULTRA" or "ON"))
end

local function disableAntiLag()
    if antiLagConn then antiLagConn:Disconnect(); antiLagConn = nil end
end

local optimizeConn = nil

local function enableOptimize()
    pcall(function()
        settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
        Lighting.GlobalShadows = false
        Lighting.Brightness    = 2
        Lighting.FogEnd        = 9e9
        Lighting.FogStart      = 9e9
        for _, fx in ipairs(Lighting:GetChildren()) do
            if fx:IsA("PostEffect") then fx.Enabled = false end
        end
    end)
    pcall(function()
        local char = LocalPlayer.Character
        for _, obj in ipairs(Workspace:GetDescendants()) do
            pcall(function()
                if char and obj:IsDescendantOf(char) then return end
                if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam")
                    or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles") then
                    obj.Enabled = false; obj:Destroy()
                elseif obj:IsA("BasePart") then
                    obj.CastShadow = false
                    obj.Material   = Enum.Material.Plastic
                    for _, child in ipairs(obj:GetChildren()) do
                        if child:IsA("Decal") or child:IsA("Texture") or child:IsA("SurfaceAppearance") then
                            child:Destroy()
                        end
                    end
                elseif obj:IsA("Sky") then
                    obj:Destroy()
                end
            end)
        end
    end)
    optimizeConn = Workspace.DescendantAdded:Connect(function(obj)
        pcall(function()
            local char = LocalPlayer.Character
            if char and obj:IsDescendantOf(char) then return end
            if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam")
                or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles") then
                obj.Enabled = false; obj:Destroy()
            elseif obj:IsA("BasePart") then
                obj.CastShadow = false
                obj.Material   = Enum.Material.Plastic
            elseif obj:IsA("Sky") then
                obj:Destroy()
            end
        end)
    end)
end

local function disableOptimize()
    if optimizeConn then optimizeConn:Disconnect(); optimizeConn = nil end
end

        end
        print("[IrishCombined]: " .. labelText .. " " .. (fenv[fenvKey] and "ON" or "OFF"))
    end)
    return btn
end

-- JOIN LOGIC
-- ============================================================
local joinCooldown  = false
local JOIN_COOLDOWN = 5

local function joinServer(placeId, jobId)
    local opened = false
    if syn and syn.request then
        pcall(function()
            syn.open_url("roblox://experiences/start?placeId=" .. tostring(placeId) .. "&gameInstanceId=" .. tostring(jobId))
            opened = true
        end)
    end
    if not opened and open_url then
        pcall(function()
            open_url("roblox://experiences/start?placeId=" .. tostring(placeId) .. "&gameInstanceId=" .. tostring(jobId))
            opened = true
        end)
    end
    if not opened then
        pcall(function() TeleportService:TeleportToPlaceInstance(placeId, jobId, LocalPlayer) end)
    end
end

local function tryJoinServer(placeId, jobId, brainrotName)
    if not placeId or not jobId then
        warn("[IrishCombined]: Cannot join - missing placeId or jobId for " .. tostring(brainrotName))
        return
    end
    if joinCooldown then
        warn("[IrishCombined]: Join on cooldown.")
        return
    end
    joinCooldown = true
    print("[IrishCombined]: Joining " .. tostring(brainrotName)
        .. " PlaceId=" .. tostring(placeId)
        .. " JobId="   .. tostring(jobId))
    joinServer(placeId, jobId)
    task.delay(JOIN_COOLDOWN, function() joinCooldown = false end)
end

local function addRow(brainrotName, subText, moneyPerSec, placeId, jobId)

    if jobId then loggedJobIds[tostring(jobId)] = true end

    if lastLogTime[brainrotName] and (now - lastLogTime[brainrotName]) < LOG_COOLDOWN then return end
    lastLogTime[brainrotName] = now

    if emptyLabel and emptyLabel.Parent then
        emptyLabel:Destroy()
        emptyLabel = nil
    end

    table.insert(serverLogs, {
        brainrotName = brainrotName,
        subText      = subText,
        moneyPerSec  = moneyPerSec,
        placeId      = placeId,
        jobId        = jobId,

    joinBtn.TextColor3      = Color3.new(1, 1, 1)

    if placeId and jobId then

        joinBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 30)

                joinBtn.Text             = "Join"
                joinBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
            end
        end)
        if fenv.AutoJoinEnabled then
            local money = tonumber(moneyPerSec) or 0

                task.delay(0.5, function() tryJoinServer(placeId, jobId, brainrotName) end)

        joinBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 35)

        local model = Workspace:FindFirstChild(tostring(brainrotName), true)
        if model then

        .. " | " .. tostring(moneyPerSec) .. "M/s"
        .. " | PlaceId=" .. tostring(placeId)
        .. " JobId=" .. tostring(jobId))

-- WORKSPACE BRAINROT SCANNER
-- ============================================================
local brainrotSet = {}
for _, name in ipairs(allBrainrots) do brainrotSet[name] = true end

local function scanWorkspaceForBrainrots()
    local now = tick()
    if now - lastScanTime < SCAN_INTERVAL then return end
    lastScanTime = now
    for _, obj in ipairs(Workspace:GetChildren()) do
        if brainrotSet[obj.Name] then
            local moneyVal = 0
            local mv = obj:FindFirstChild("MoneyPerSec")
                    or obj:FindFirstChild("Money")
                    or obj:FindFirstChild("MPS")
                    or obj:FindFirstChild("Income")
            if mv and (mv:IsA("NumberValue") or mv:IsA("IntValue")) then
                moneyVal = mv.Value
            end
            local jobId   = obj:GetAttribute("JobId")
                         or (obj:FindFirstChild("JobId") and obj:FindFirstChild("JobId").Value) or nil
            local placeId = obj:GetAttribute("PlaceId")
                         or (obj:FindFirstChild("PlaceId") and obj:FindFirstChild("PlaceId").Value) or nil
            local money   = tonumber(moneyVal) or 0
            if money >= fenv.MinMoney or not fenv.AutoJoinEnabled then
                addRow(obj.Name, "Found in Workspace", money, placeId, jobId)
            end
        end
    end
end

Workspace.ChildAdded:Connect(function(obj)
    task.wait(0.5)
    if brainrotSet[obj.Name] then
        local moneyVal = 0
        local mv = obj:FindFirstChild("MoneyPerSec")
                or obj:FindFirstChild("Money")
                or obj:FindFirstChild("MPS")
                or obj:FindFirstChild("Income")
        if mv and (mv:IsA("NumberValue") or mv:IsA("IntValue")) then
            moneyVal = mv.Value
        end
        local jobId   = obj:GetAttribute("JobId")
                     or (obj:FindFirstChild("JobId") and obj:FindFirstChild("JobId").Value) or nil
        local placeId = obj:GetAttribute("PlaceId")
                     or (obj:FindFirstChild("PlaceId") and obj:FindFirstChild("PlaceId").Value) or nil
        local money   = tonumber(moneyVal) or 0
        if money >= fenv.MinMoney or not fenv.AutoJoinEnabled then
            addRow(obj.Name, "Appeared in Workspace", money, placeId, jobId)
        end
    end
end)

task.spawn(function()
    while true do
        task.wait(SCAN_INTERVAL)
        scanWorkspaceForBrainrots()
    end
end)

            and Color3.fromRGB(0, 110, 0) or Color3.fromRGB(30, 30, 30)
    end)

        btn.Visible = query == "" or name:lower():find(query, 1, true) ~= nil
    end
end)

            brainrotButtonRefs[name].BackgroundColor3 = Color3.fromRGB(0, 110, 0)
        end
    end
end)

        if brainrotButtonRefs[name] then
            brainrotButtonRefs[name].BackgroundColor3 = Color3.fromRGB(30, 30, 30)
        end
    end
end)
